  246: def get_implmap_scope(row):
  247:     scope = row_ref(getattr(row, "ImportScope", None))
  248:     return row_name(scope, "Name") if scope is not None else ""

  251: def write_csv(path, rows):
  252:     if not rows:
  253:         with open(path, "w", encoding="utf-8", newline="") as handle:
  254:             handle.write("")
  255:         return
  256:     columns = []
  257:     for row in rows:
  258:         for key in row:
  259:             if key not in columns:
  260:                 columns.append(key)
  261:     with open(path, "w", encoding="utf-8", newline="") as handle:
  262:         writer = csv.DictWriter(handle, fieldnames=columns, extrasaction="ignore")
  263:         writer.writeheader()
  264:         writer.writerows(rows)

  267: def scan_one(path, all_calls, all_declarations, all_dynamic, errors, counters):
  268:     basename = os.path.basename(path)
  269:     pe = None
  270:     try:
  271:         pe = dnfile.dnPE(path)
  272:         if not getattr(pe, "net", None) or not getattr(pe.net, "mdtables", None):
  273:             counters["non_managed_or_no_metadata"] += 1
  274:             return
  275:         md = pe.net.mdtables
  276:         methods = getattr(getattr(md, "MethodDef", None), "rows", []) or []
  277:         implmaps = getattr(getattr(md, "ImplMap", None), "rows", []) or []
  278:         memberrefs = getattr(getattr(md, "MemberRef", None), "rows", []) or []
  279:         methodspecs = getattr(getattr(md, "MethodSpec", None), "rows", []) or []
  280:         current_assembly = assembly_identity(pe, path)
  281:         method_rid_by_obj = {id(row): rid for rid, row in enumerate(methods, 1)}
  282:         owners = build_owner_map(md)
  283: 
  284:         implmap_by_rid = {}
  285:         counters["implmap_rows_seen"] += len(implmaps)
  286:         for implmap in implmaps:
  287:             try:
  288:                 forwarded = row_ref(getattr(implmap, "MemberForwarded", None))
  289:                 rid = method_rid_by_obj.get(id(forwarded), 0)
  290:                 if not rid:
  291:                     raise ValueError("MemberForwarded did not resolve to a MethodDef row in this module")
  292:                 if rid in implmap_by_rid:
  293:                     counters["implmap_duplicate_method_rows"] += 1
  294:                     raise ValueError(f"multiple ImplMap rows for MethodDef RID {rid}")
  295:                 implmap_by_rid[rid] = {
  296:                     "dll": get_implmap_scope(implmap),
  297:                     "entrypoint": row_name(implmap, "ImportName") or row_name(forwarded),
  298:                     "managed_method": row_name(forwarded),
  299:                     "managed_type": owners.get(rid, ""),
  300:                     "signature_hex": signature_hex(forwarded),
  301:                     "assembly_identity": current_assembly,
  302:                     "method_rid": rid,
  303:                 }
  304:             except Exception as exc:
  305:                 counters["implmap_resolution_errors"] += 1
  306:                 errors.append({"assembly": basename, "stage": "ImplMap", "detail": text(exc)})
  307: 
  308:         flagged_rids = set()
  309:         for rid, method_row in enumerate(methods, 1):
  310:             try:
  311:                 if method_has_pinvoke_flag(method_row):
  312:                     flagged_rids.add(rid)
  313:             except Exception as exc:
  314:                 counters["pinvoke_flag_read_errors"] += 1
  315:                 errors.append({"assembly": basename, "stage": "MethodDefFlags", "detail": f"RID {rid}: {exc}"})
  316:         missing_implmap = flagged_rids - set(implmap_by_rid)
  317:         orphan_implmap = set(implmap_by_rid) - flagged_rids
  318:         counters["pinvoke_flagged_methoddefs"] += len(flagged_rids)
  319:         counters["pinvoke_declarations_resolved"] += len(implmap_by_rid)
  320:         counters["pinvoke_missing_implmap"] += len(missing_implmap)
  321:         counters["implmap_orphan_rows"] += len(orphan_implmap)
  322:         for rid in sorted(missing_implmap):
  323:             errors.append({"assembly": basename, "stage": "ImplMapCoverage", "detail": f"MethodDef RID {rid} has mdPinvokeImpl but no resolved ImplMap"})
  324:         for rid in sorted(orphan_implmap):
  325:             errors.append({"assembly": basename, "stage": "ImplMapCoverage", "detail": f"ImplMap resolved to MethodDef RID {rid} without mdPinvokeImpl flag"})
  326: 
  327:         declarations = []
  328:         for rid, declaration in implmap_by_rid.items():
  329:             declarations.append({
  330:                 "assembly": basename,
  331:                 "assembly_identity": current_assembly,
  332:                 "managed_type": declaration["managed_type"],
  333:                 "managed_method": declaration["managed_method"],
  334:                 "method_rid": rid,
  335:                 "dll": declaration["dll"],
  336:                 "entrypoint": declaration["entrypoint"],
  337:                 "signature_hex": declaration["signature_hex"],
  338:             })
  339:         all_declarations.extend(declarations)
  340: 
  341:         counters["managed_files_scanned"] += 1
  342:         counters["methoddefs_seen"] += len(methods)
  343:         counters["method_bodies_attempted"] += sum(1 for row in methods if getattr(row, "Rva", 0))
  344: 
  345:         for caller_rid, caller in enumerate(methods, 1):
  346:             rva = getattr(caller, "Rva", 0) or 0
  347:             if not rva:
  348:                 continue
  349:             try:
  350:                 body = CilMethodBody(CilMethodBodyReaderBytes(pe.get_data(rva)))
  351:                 instructions = getattr(body, "instructions", []) or []
  352:                 counters["method_bodies_parsed"] += 1
  353:             except Exception as exc:
  354:                 counters["method_body_parse_errors"] += 1
  355:                 errors.append({"assembly": basename, "stage": "ILParse", "detail": f"MethodDef RID {caller_rid}: {type(exc).__name__}: {exc}"})
  356:                 continue
  357: 
  358:             caller_name = row_name(caller)
  359:             caller_type = owners.get(caller_rid, "")
  360:             for instruction in instructions:
  361:                 try:
  362:                     opcode = text(instruction.opcode.name).lower()
  363:                 except Exception:
  364:                     opcode = text(getattr(instruction, "opcode", "")).lower()
  365:                 if opcode not in CALL_OPS:
  366:                     continue
  367:                 token = token_value(getattr(instruction, "operand", None))
  368:                 table = table_name(token) if token is not None else None
  369:                 rid = token_rid(token) if token is not None else None
  370:                 target_name = target_type = target_assembly = target_dll = target_entry = target_kind = ""
  371:                 target_sig = ""
  372:                 if table == 0x06 and rid and rid <= len(methods):
  373:                     target = methods[rid - 1]
  374:                     target_name = row_name(target)
  375:                     target_type = owners.get(rid, "")
  376:                     target_assembly = current_assembly
  377:                     target_sig = signature_hex(target)
  378:                     target_kind = "MethodDef"
  379:                     declaration = implmap_by_rid.get(rid)
  380:                     if declaration:
  381:                         target_dll = declaration["dll"]
  382:                         target_entry = declaration["entrypoint"]
  383:                 elif table == 0x0A and rid and rid <= len(memberrefs):
  384:                     target = memberrefs[rid - 1]
  385:                     target_name = row_name(target)
  386:                     target_assembly, target_type, _, target_sig = method_identity(target, current_assembly, owners, method_rid_by_obj)
  387:                     target_kind = "MemberRef"
  388:                 elif table == 0x2B and rid and rid <= len(methodspecs):
  389:                     spec = methodspecs[rid - 1]
  390:                     base = row_ref(getattr(spec, "Method", None))
  391:                     if base is not None:
  392:                         target_name = row_name(base)
  393:                         target_assembly, target_type, _, target_sig = method_identity(base, current_assembly, owners, method_rid_by_obj)
  394:                     target_kind = "MethodSpec"
  395:                 row = {
  396:                     "assembly": basename,
  397:                     "assembly_identity": current_assembly,
  398:                     "caller_type": caller_type,
  399:                     "caller_method": caller_name,
  400:                     "caller_method_rid": caller_rid,
  401:                     "il_offset": getattr(instruction, "offset", ""),
  402:                     "opcode": opcode,
  403:                     "target_token": f"0x{token:08X}" if token is not None else "",
  404:                     "target_table": f"0x{table:02X}" if table is not None else "",
  405:                     "target_kind": target_kind,
  406:                     "target_assembly": target_assembly,
  407:                     "target_type": target_type,
  408:                     "target_method": target_name,
  409:                     "target_signature_hex": target_sig,
  410:                     "pinvoke_dll": target_dll,
  411:                     "pinvoke_entrypoint": target_entry,
  412:                 }
  413:                 all_calls.append(row)
  414:                 if target_dll or target_name.lower() in DYNAMIC_NAMES:
  415:                     all_declarations.append({
  416:                         "assembly": basename, "assembly_identity": current_assembly,
  417:                         "managed_type": caller_type, "managed_method": caller_name,
  418:                         "method_rid": caller_rid, "dll": target_dll,
  419:                         "entrypoint": target_entry, "signature_hex": "",
  420:                         "record_kind": "DIRECT_CALLSITE", "target_method": target_name,
  421:                         "target_type": target_type,
  422:                     })
  423:                 if target_name.lower() in DYNAMIC_NAMES:
  424:                     all_dynamic.append(row)
  425:     except Exception as exc:
  426:         counters["file_errors"] += 1
  427:         errors.append({"assembly": basename, "stage": "File", "detail": f"{type(exc).__name__}: {exc}"})
  428:         print(f"[ERROR] {path}: {type(exc).__name__}: {exc}", file=sys.stderr)
  429:     finally:
  430:         if pe is not None:
  431:             try:
  432:                 pe.close()
  433:             except Exception:
  434:                 pass

  437: def enrich_cross_assembly_calls(calls, declarations, counters):
  438:     index = defaultdict(list)
  439:     for declaration in declarations:
  440:         if declaration.get("record_kind") == "DIRECT_CALLSITE":
  441:             continue
  442:         key = (
  443:             declaration.get("assembly_identity", "").casefold(),
  444:             declaration.get("managed_type", "").casefold(),
  445:             declaration.get("managed_method", "").casefold(),
  446:             declaration.get("signature_hex", "").lower(),
  447:         )
  448:         if all(key):
  449:             index[key].append(declaration)
  450: 
  451:     for row in calls:
  452:         if row.get("target_kind") not in ("MemberRef", "MethodSpec"):
  453:             continue
  454:         key = (
  455:             row.get("target_assembly", "").casefold(),
  456:             row.get("target_type", "").casefold(),
  457:             row.get("target_method", "").casefold(),
  458:             row.get("target_signature_hex", "").lower(),
  459:         )
  460:         if not all(key):
  461:             counters["memberref_identity_incomplete"] += 1
  462:             continue
  463:         matches = index.get(key, [])
  464:         if len(matches) == 1:
  465:             match = matches[0]
  466:             row["pinvoke_dll"] = match.get("dll", "")
  467:             row["pinvoke_entrypoint"] = match.get("entrypoint", "")
  468:             row["target_kind"] += "_PInvokeExactMatch"
  469:             counters["memberref_pinvoke_exact_matches"] += 1
  470:         elif len(matches) > 1:
  471:             row["target_kind"] += "_PInvokeAmbiguous"
  472:             counters["memberref_pinvoke_ambiguous"] += 1
  473:         else:
  474:             # Most MemberRefs are ordinary managed/native API references, not P/Invokes.
  475:             counters["memberref_no_pinvoke_match"] += 1
